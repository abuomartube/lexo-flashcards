<script>
// ===============================
// LEXO Swipe System
// ===============================

const card = document.getElementById('card');

let startX = 0;
let currentX = 0;
let isDragging = false;
let threshold = 70;

// Demo words
let words = ["April", "Ability", "Future", "Growth"];
let index = 0;

// ===============================
// Update Card
// ===============================
function updateCard() {
  card.textContent = words[index];
}

// ===============================
// Swipe Start
// ===============================
card.addEventListener('touchstart', (e) => {
  startX = e.touches[0].clientX;
  isDragging = true;
});

// ===============================
// Swipe Move
// ===============================
card.addEventListener('touchmove', (e) => {
  if (!isDragging) return;

  currentX = e.touches[0].clientX;
  let diff = currentX - startX;

  card.style.transform = `translateX(${diff}px)`;

  // Glow feedback
  if (diff > 0) {
    card.style.background = 'rgba(0,255,150,0.1)';
  } else {
    card.style.background = 'rgba(255,80,80,0.1)';
  }
});

// ===============================
// Swipe End
// ===============================
card.addEventListener('touchend', () => {
  if (!isDragging) return;

  let diff = currentX - startX;

  card.style.transition = 'transform 0.3s ease';
  card.style.transform = 'translateX(0)';
  card.style.background = '';

  if (diff > threshold) {
    nextCard();
  } else if (diff < -threshold) {
    prevCard();
  }

  isDragging = false;
});

// ===============================
// Buttons
// ===============================
document.getElementById('next').addEventListener('click', nextCard);
document.getElementById('prev').addEventListener('click', prevCard);

// ===============================
// Logic
// ===============================
function nextCard() {
  index = (index + 1) % words.length;
  updateCard();
}

function prevCard() {
  index = (index - 1 + words.length) % words.length;
  updateCard();
}

// Init
updateCard();

</script>
